Program 8: Swapping Two Numbers

public class SwapNumbers {
    public static void main(String[] args) {
        int a = 10;
        int b = 20;
        int temp;

        System.out.println("Before Swapping:");
        System.out.println("a = " + a);
        System.out.println("b = " + b);

        temp = a;
        a = b;
        b = temp;

        System.out.println("After Swapping:");
        System.out.println("a = " + a);
        System.out.println("b = " + b);
    }
}

Output:

Before Swapping:
a = 10
b = 20
After Swapping:
a = 20
b = 10

Program 12: Even or Odd

import java.util.Scanner;

public class EvenOddCheck {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter a number: ");
        int num = sc.nextInt();

        if (num % 2 == 0) {
            System.out.println(num + " is Even");
        } else {
            System.out.println(num + " is Odd");
        }

        sc.close();
    }
}

Sample Output:

Enter a number: 8
8 is Even

Program 13: Largest of Two Numbers

import java.util.Scanner;

public class LargestOfTwo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter first number: ");
        int a = sc.nextInt();

        System.out.print("Enter second number: ");
        int b = sc.nextInt();

        if (a > b) {
            System.out.println(a + " is the largest number");
        } else if (b > a) {
            System.out.println(b + " is the largest number");
        } else {
            System.out.println("Both numbers are equal");
        }

        sc.close();
    }
}

Sample Output:

Enter first number: 10
Enter second number: 20
20 is the largest number

Program 14: Java Reserved Keywords

public class KeywordDemo {
    public static void main(String[] args) {
        int age = 20;
        float salary = 25000.50f;
        char grade = 'A';

        System.out.println("Age = " + age);
        System.out.println("Salary = " + salary);
        System.out.println("Grade = " + grade);

        System.out.println("\nExplanation:");
        System.out.println("Keywords like 'class', 'if', 'public' are reserved by Java.");
        System.out.println("They have predefined meanings in the language syntax.");
        System.out.println("Therefore, they cannot be used as variable names.");
    }
}

Output:

Age = 20
Salary = 25000.5
Grade = A

Explanation:
Keywords like 'class', 'if', 'public' are reserved by Java.
They have predefined meanings in the language syntax.
Therefore, they cannot be used as variable names.
